        cd "$srcdir/${pkgname}"

}

prepare() {
        cd "$srcdir/${pkgname}"
        export RUSTUP_TOOLCHAIN=stable
        cargo fetch --locked --target "$(rustc -vV | sed -n 's|host: ||p')"
}

build() {
   cd "$srcdir/${pkgname}"
    export RUSTUP_TOOLCHAIN=stable
    export CARGO_TARGET_DIR=target
    make
  }

package() {
   cd "$srcdir/${pkgname}"
   make install DESTDIR="$pkgdir" PERFIX=/usr LIBEXECDIR=/usr/lib
}
